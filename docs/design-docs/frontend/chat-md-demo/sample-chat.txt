我看了一下，问题出在 `apps/desktop/src/renderer/components/messages/MarkdownProse.tsx:118`，流式期间每帧都会重新解析最后一块。改动如下：

根因是 `splitMarkdownBlocks()` 在 streaming 时没有复用上一次的切块结果，所以每来一个 token，React 都要重新 reconcile 整条回复。回复越长越明显：一条 30KB 的回答，每次更新要花 34ms，已经超过一帧的预算（16.7ms），滚动和输入都会跟着发涩。

## 改了什么

1. 在 `markdown-blocks.ts` 里缓存上一次的切块结果；
2. 调整 `rehypeHighlight` 的 `detect` 参数，只在围栏闭合后识别语言；
3. 新增测试 `apps/desktop/src/renderer/test/markdown-blocks.test.tsx`。

```ts title="apps/desktop/src/renderer/components/messages/markdown-blocks.ts"
export function splitMarkdownBlocks(content: string, previous?: MarkdownBlockSplit): MarkdownBlockSplit {
  if (previous && previous.content === content) return previous;
  const blocks: MarkdownBlock[] = [];
  // 只重新切最后一块：前面的块在流式期间不会变
  return { content, blocks };
}
```

```ts title="components/markdown-editor/core/config.ts"
import { EditorState, Compartment } from "@codemirror/state";
import { EditorView, keymap, lineNumbers } from "@codemirror/view";
import { markdown, markdownKeymap } from "@codemirror/lang-markdown";
import { history, historyKeymap, defaultKeymap } from "@codemirror/commands";
import { search, searchKeymap } from "@codemirror/search";

export function createEditor(parent: HTMLElement, doc: string) {
  const mode = new Compartment();
  const readOnly = new Compartment();
  const state = EditorState.create({
    doc,
    extensions: [
      history(),
      search({ top: true }),
      markdown(),
      mode.of([]),
      readOnly.of(EditorState.readOnly.of(false)),
      keymap.of([...markdownKeymap, ...defaultKeymap, ...historyKeymap, ...searchKeymap]),
      EditorView.lineWrapping,
    ],
  });
  const view = new EditorView({ state, parent });
  return {
    view,
    setSource(source: boolean) {
      view.dispatch({ effects: mode.reconfigure(source ? lineNumbers() : []) });
    },
    setReadOnly(value: boolean) {
      view.dispatch({ effects: readOnly.reconfigure(EditorState.readOnly.of(value)) });
    },
    destroy() {
      view.destroy();
    },
  };
}
```
